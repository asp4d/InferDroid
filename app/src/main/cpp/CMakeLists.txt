cmake_minimum_required(VERSION 3.22.1)
project(inferdroid_jni LANGUAGES CXX)

if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Milestone 1 supports arm64-v8a only.")
endif()
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(RUNTIME_DIR "${PROJECT_ROOT}/native/artifacts/arm64-v8a")
foreach(LIBRARY libinferdroid_litertlm.so libLiteRtDispatch_GoogleTensor.so libGemmaModelConstraintProvider.so)
    if(NOT EXISTS "${RUNTIME_DIR}/${LIBRARY}")
        message(FATAL_ERROR "Missing ${RUNTIME_DIR}/${LIBRARY}. Run scripts/build-native.sh; see README.md.")
    endif()
endforeach()

add_library(inferdroid_litertlm SHARED IMPORTED)
set_target_properties(inferdroid_litertlm PROPERTIES
    IMPORTED_LOCATION "${RUNTIME_DIR}/libinferdroid_litertlm.so")
add_library(inferdroid_jni SHARED native_inference.cpp)
target_include_directories(inferdroid_jni PRIVATE "${PROJECT_ROOT}/native")
target_compile_features(inferdroid_jni PRIVATE cxx_std_17)
target_compile_options(inferdroid_jni PRIVATE -Wall -Wextra -Werror)
target_link_options(inferdroid_jni PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384)
target_link_libraries(inferdroid_jni PRIVATE inferdroid_litertlm log)

set(SPEECH_DIR "${PROJECT_ROOT}/native/artifacts/speech")
foreach(LIBRARY libsherpa-onnx-c-api.so libonnxruntime.so)
    if(NOT EXISTS "${SPEECH_DIR}/arm64-v8a/${LIBRARY}")
        message(FATAL_ERROR "Missing speech runtime ${LIBRARY}. Run scripts/build-speech.sh; see README.md.")
    endif()
endforeach()
if(NOT EXISTS "${SPEECH_DIR}/include/sherpa-onnx/c-api/c-api.h")
    message(FATAL_ERROR "Missing pinned speech C API header. Run scripts/build-speech.sh.")
endif()
foreach(NOTICE sherpa-onnx-LICENSE.txt onnxruntime-LICENSE.txt onnxruntime-ThirdPartyNotices.txt dependencies.txt whisper-LICENSE.txt supertonic-model-LICENSE.txt)
    if(NOT EXISTS "${PROJECT_ROOT}/app/src/main/assets/third_party/generated/speech/${NOTICE}")
        message(FATAL_ERROR "Missing speech runtime notice ${NOTICE}. Run scripts/build-speech.sh.")
    endif()
endforeach()
add_library(sherpa_onnx SHARED IMPORTED)
set_target_properties(sherpa_onnx PROPERTIES IMPORTED_LOCATION "${SPEECH_DIR}/arm64-v8a/libsherpa-onnx-c-api.so")
add_library(onnxruntime SHARED IMPORTED)
set_target_properties(onnxruntime PROPERTIES IMPORTED_LOCATION "${SPEECH_DIR}/arm64-v8a/libonnxruntime.so")
add_library(inferdroid_speech_jni SHARED native_speech.cpp native_tts.cpp)
target_include_directories(inferdroid_speech_jni PRIVATE "${SPEECH_DIR}/include")
target_compile_features(inferdroid_speech_jni PRIVATE cxx_std_17)
target_compile_options(inferdroid_speech_jni PRIVATE -Wall -Wextra -Werror)
target_link_options(inferdroid_speech_jni PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384)
target_link_libraries(inferdroid_speech_jni PRIVATE sherpa_onnx onnxruntime log)
