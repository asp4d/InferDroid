cmake_minimum_required(VERSION 3.22.1)
project(inferdroid_research_images LANGUAGES CXX)
get_filename_component(INFERDROID_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_subdirectory("${INFERDROID_ROOT}/app/src/main/cpp" base-app)
add_library(inferdroid_images_jni SHARED src/cpp/images_jni.cc)
target_include_directories(inferdroid_images_jni PRIVATE "${INFERDROID_ROOT}/.local/tensor-g5/cache/device-runtime-2.3.0-r2/include")
target_compile_features(inferdroid_images_jni PRIVATE cxx_std_17)
target_compile_options(inferdroid_images_jni PRIVATE -Wall -Wextra -Werror)
target_link_options(inferdroid_images_jni PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384)
target_link_libraries(inferdroid_images_jni PRIVATE dl log)
